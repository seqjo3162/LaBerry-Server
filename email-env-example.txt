# PowerShell / .env пример для отправки кодов подтверждения email
# Команда вызывается сервером после генерации 6-значного кода.

LB_EMAIL_SEND_COMMAND=powershell.exe -NoProfile -ExecutionPolicy Bypass -File .\send-email-code.ps1

LB_SMTP_HOST=smtp.example.com
LB_SMTP_PORT=587
LB_SMTP_SSL=true
LB_SMTP_USER=bot@example.com
LB_SMTP_PASS=your_password_or_app_password
LB_SMTP_FROM=bot@example.com

# Только для локальной отладки. В продакшене выключить.
LB_DEBUG_EMAIL_CODES=false
